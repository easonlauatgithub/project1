class playerDAO {

Mapper mapper = getMapping();

String getPlayerName(){
 mapper.select("SEARCH_PLAYER_NAME");
}

String getPlayerPosition(){
 mapper.select("SEARCH_PLAYER_POSITION");
}

String getPlayerNumber(){
 mapper.select("SEARCH_PLAYER_NUMBER").toString;
}

String getPlayerNationality(){
 mapper.select("SEARCH_PLAYER_NATIONALITY");
}

}
